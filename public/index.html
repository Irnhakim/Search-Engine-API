            GET
            /api/search
            Web Search
          

          

            

              
ParameterTypeRequiredDescription

              
qstringYesSearch query

              
pagenumberNoResult page (default: 1)

              
regionstringNoRegion code (default: wt-wt)

              
safeSearchstringNomoderate / strict / off

            

            

              
Example Request 

              
curl -H "X-Api-Key: YOUR_KEY" \
  "http://localhost:3000/api/search?q=quantum+computing"

            POST
            /api/scrape
            Scrape URL
          

          

            

              
FieldTypeRequiredDescription

              
urlstringYesTarget URL to scrape

              
maxLengthnumberNoMax content chars (default: 5000)

            

            

              
curl -X POST -H "X-Api-Key: YOUR_KEY" \

  "http://localhost:3000/api/scrape"

            

          

        


        
        

          

            POST
            /api/research
            🧠 Deep Research (AI-Ready)
          

          

            

              
FieldTypeRequiredDescription

              
topicstringYesResearch topic or question

              
maxResultsnumberNoMax search results (default: 10)

              
maxScrapenumberNoMax pages to scrape (default: 5)

              
scrapeContentboolNoScrape page content (default: true)

              
regionstringNoSearch region code

            

            

              
Example — AI Model Integration

              
// 1. Call deep research
const response = await fetch('/api/research', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'X-Api-Key': 'YOUR_KEY'
  },
  body: JSON.stringify({
    topic: 'What is the latest news about GPT-5?',
    maxResults: 15,
    maxScrape: 5,
    scrapeContent: true
  })
});

const { data } = await response.json();

// 2. Feed aiContext to your AI model
const aiResponse = await openai.chat.completions.create({
  model: 'gpt-4',
  messages: [
    { role: 'system', content: 'You are a research assistant.' },
    { role: 'user', content: data.aiContext }, // ← Full research context
    { role: 'user', content: 'Summarize the key findings.' }
  ]
});
